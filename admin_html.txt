    <script>
        if (!localStorage.getItem('token')) {
            window.location.href = '/login.html';
        }

        let todosOsLeads = [];
        let todasAsTags = [];
        let todosOsClientes = [];
        let leadIdAtual = null;

        // --- FUNÇÕES UTILITÁRIAS ---
        function escapeHtml(text) {
            if (!text) return '';
            const div = document.createElement('div');
            div.textContent = text;
            return div.innerHTML;
        }

        // --- CARREGAMENTO DE DADOS ---

        async function carregarClientesSelect() {
            const token = localStorage.getItem('token');
            try {
                const response = await fetch('/api/clientes', {
                    headers: { 'Authorization': `Bearer ${token}` }
                });
                if (response.ok) {
                    todosOsClientes = await response.json();
                }
            } catch (error) {
                console.error('Erro ao carregar clientes:', error);
            }
        }

        async function carregarTags() {
            const token = localStorage.getItem('token');
            try {
                const response = await fetch('/api/tags', {
                    headers: { 'Authorization': `Bearer ${token}` }
                });
                if (response.ok) {
                    todasAsTags = await response.json();
                    popularSelectTags();
                    renderizarListaTagsDisponiveis();
                }
            } catch (error) {
                console.error('Erro ao carregar tags:', error);
            }
        }

        async function carregarTotalClientes() {
            const token = localStorage.getItem('token');
            try {
                const response = await fetch('/api/metricas', {
                    headers: { 'Authorization': `Bearer ${token}` }
                });
                if (response.ok) {
                    const data = await response.json();
                    document.getElementById('totalClientes').textContent = data.totalClientes || 0;
                }
            } catch (error) {
                console.error('Erro ao carregar total de clientes:', error);
            }
        }

        // --- CARREGAMENTO PRINCIPAL (ORDEM CORRETA) ---
        async function carregarTudo() {
            // 1. Carrega clientes PRIMEIRO (para a coluna "Cliente" funcionar)
            await carregarClientesSelect();
            // 2. Carrega tags
            await carregarTags();
            // 3. Carrega os leads (agora com clientes já disponíveis)
            await carregarLeads();
            // 4. Carrega o total de clientes (card)
            await carregarTotalClientes();
        }

        async function carregarLeads() {
            const token = localStorage.getItem('token');
            try {
                const response = await fetch('/api/leads', {
                    headers: { 'Authorization': `Bearer ${token}` }
                });
                if (!response.ok) throw new Error('Falha ao buscar leads');
                const leads = await response.json();
                todosOsLeads = leads;

                const total = leads.length;
                const pendentes = leads.filter(l => l.status === 'pendente' || !l.status).length;
                const atendidos = leads.filter(l => l.status === 'atendido').length;

                document.getElementById('totalLeads').textContent = total;
                document.getElementById('leadsPendentes').textContent = pendentes;
                document.getElementById('leadsAtendidos').textContent = atendidos;

                filtrarLeads();
            } catch (error) {
                console.error('Erro:', error);
                alert('Erro ao carregar dados. Verifique o console.');
            }
        }

        function popularSelectTags() {
            const select = document.getElementById('filtrarTag');
            const valorAtual = select.value;
            select.innerHTML = '<option value="">Todas as Tags</option>';
            todasAsTags.forEach(tag => {
                const option = document.createElement('option');
                option.value = tag.id;
                option.textContent = tag.nome;
                select.appendChild(option);
            });
            select.value = valorAtual;
        }

        function renderizarListaTagsDisponiveis() {
            const container = document.getElementById('listaTagsDisponiveis');
            container.innerHTML = '';
            if (todasAsTags.length === 0) {
                container.innerHTML = '<p class="text-slate-500 text-sm">Nenhuma tag cadastrada.</p>';
                return;
            }
            todasAsTags.forEach(tag => {
                const span = document.createElement('span');
                span.className = 'bg-slate-800 text-slate-300 px-3 py-1 rounded text-sm border border-slate-700';
                span.textContent = tag.nome;
                container.appendChild(span);
            });
        }

        // --- RENDERIZAÇÃO DA TABELA ---

        function renderizarTabela(leads) {
            const tbody = document.getElementById('tabelaLeads');
            tbody.innerHTML = '';

            if (leads.length === 0) {
                tbody.innerHTML = `<tr><td colspan="10" class="p-4 text-center text-slate-400">Nenhum lead encontrado.</td></tr>`;
                return;
            }

            leads.forEach(lead => {
                const statusColor = lead.status === 'atendido' 
                    ? 'text-green-400 bg-green-400/10' 
                    : 'text-yellow-400 bg-yellow-400/10';
                
                const tagsHtml = (lead.tags || []).map(t => 
                    `<span class="bg-orange-500/20 text-orange-400 px-2 py-0.5 rounded text-xs font-medium">${escapeHtml(t.nome)}</span>`
                ).join(' ');

                const cliente = todosOsClientes.find(c => c.id == lead.cliente_id);
                const clienteNome = cliente ? cliente.nome : '-';

                const tr = document.createElement('tr');
                tr.className = 'hover:bg-slate-800/50 transition-colors';
                tr.innerHTML = `
                    <td class="p-4 text-slate-400">${lead.id}</td>
                    <td class="p-4 font-medium">${escapeHtml(lead.nome)}</td>
                    <td class="p-4 text-blue-400">${escapeHtml(lead.email)}</td>
                    <td class="p-4 text-slate-300">${escapeHtml(lead.telefone) || '-'}</td>
                    <td class="p-4 text-slate-300">${escapeHtml(lead.empresa) || '-'}</td>
                    <td class="p-4 text-slate-300">${escapeHtml(lead.origem) || '-'}</td>
                    <td class="p-4 text-slate-300">${escapeHtml(clienteNome)}</td>
                    <td class="p-4"><div class="flex flex-wrap gap-1">${tagsHtml || '<span class="text-slate-500 text-xs">Sem tags</span>'}</div></td>
                    <td class="p-4"><span class="px-2 py-1 rounded text-xs font-bold ${statusColor}">${lead.status || 'pendente'}</span></td>
                    <td class="p-4 flex gap-2 flex-wrap">
                        <button onclick="abrirEdicaoLead(${lead.id})" class="bg-cyan-600 hover:bg-cyan-700 text-white px-3 py-1 rounded text-sm transition-colors">Editar</button>
                        <button onclick="abrirGerenciarTags(${lead.id})" class="bg-purple-600 hover:bg-purple-700 text-white px-3 py-1 rounded text-sm transition-colors">Tags</button>
                        <button onclick="abrirHistorico(${lead.id})" class="bg-blue-600 hover:bg-blue-700 text-white px-3 py-1 rounded text-sm transition-colors">Histórico</button>
                        <button onclick="atenderLead(${lead.id})" class="bg-green-600 hover:bg-green-700 text-white px-3 py-1 rounded text-sm transition-colors">Atender</button>
                        <button onclick="excluirLead(${lead.id})" class="bg-red-500 hover:bg-red-600 text-white px-3 py-1 rounded text-sm transition-colors">Excluir</button>
                    </td>
                `;
                tbody.appendChild(tr);
            });
        }

        // --- FILTROS ---

        function filtrarLeads() {
            const termoBusca = document.getElementById('buscarLead').value.toLowerCase();
            const statusFiltro = document.getElementById('filtrarStatus').value;
            const tagFiltro = document.getElementById('filtrarTag').value;

            const leadsFiltrados = todosOsLeads.filter(lead => {
                const matchBusca = lead.nome.toLowerCase().includes(termoBusca) || 
                                   lead.email.toLowerCase().includes(termoBusca);
                const matchStatus = statusFiltro === '' || 
                                    (lead.status || 'pendente') === statusFiltro;
                const matchTag = tagFiltro === '' || 
                                 (lead.tags || []).some(t => t.id == tagFiltro);
                return matchBusca && matchStatus && matchTag;
            });

            renderizarTabela(leadsFiltrados);
        }

        document.getElementById('buscarLead').addEventListener('input', filtrarLeads);
        document.getElementById('btnBuscar').addEventListener('click', filtrarLeads);
        document.getElementById('filtrarStatus').addEventListener('change', filtrarLeads);
        document.getElementById('filtrarTag').addEventListener('change', filtrarLeads);
        document.getElementById('limparFiltros').addEventListener('click', () => {
            document.getElementById('buscarLead').value = '';
            document.getElementById('filtrarStatus').value = '';
            document.getElementById('filtrarTag').value = '';
            filtrarLeads();
        });

        // --- EDIÇÃO DE LEAD ---

        function abrirEdicaoLead(id) {
            const lead = todosOsLeads.find(l => l.id === id);
            if (!lead) return;

            document.getElementById('editLeadId').value = lead.id;
            document.getElementById('editLeadStatus').value = lead.status || 'pendente';
            
            const selectCliente = document.getElementById('editLeadCliente');
            selectCliente.innerHTML = '<option value="">Nenhum</option>';
            todosOsClientes.forEach(c => {
                const option = document.createElement('option');
                option.value = c.id;
                option.textContent = c.nome;
                if (c.id == lead.cliente_id) option.selected = true;
                selectCliente.appendChild(option);
            });

            document.getElementById('modalEdicaoLead').classList.remove('hidden');
        }

        document.getElementById('fecharModalEdicaoLead').addEventListener('click', () => {
            document.getElementById('modalEdicaoLead').classList.add('hidden');
        });

        document.getElementById('modalEdicaoLead').addEventListener('click', (e) => {
            if (e.target.id === 'modalEdicaoLead') {
                document.getElementById('modalEdicaoLead').classList.add('hidden');
            }
        });

        document.getElementById('formEdicaoLead').addEventListener('submit', async (e) => {
            e.preventDefault();
            const id = document.getElementById('editLeadId').value;
            const status = document.getElementById('editLeadStatus').value;
            const cliente_id = document.getElementById('editLeadCliente').value || null;

            const token = localStorage.getItem('token');
            try {
                const response = await fetch(`/api/leads/${id}`, {
                    method: 'PUT',
                    headers: {
                        'Content-Type': 'application/json',
                        'Authorization': `Bearer ${token}`
                    },
                    body: JSON.stringify({ status, cliente_id })
                });
                if (!response.ok) throw new Error('Falha ao atualizar lead');
                alert('Lead atualizado!');
                document.getElementById('modalEdicaoLead').classList.add('hidden');
                await carregarLeads();
            } catch (error) {
                console.error('Erro:', error);
                alert('Erro ao atualizar lead.');
            }
        });

        // --- LÓGICA DE TAGS ---

        async function abrirGerenciarTags(id) {
            leadIdAtual = id;
            const lead = todosOsLeads.find(l => l.id === id);
            const leadTags = lead.tags || [];

            document.getElementById('modalTags').classList.remove('hidden');
            
            const container = document.getElementById('listaTagsDisponiveis');
            container.innerHTML = '';

            if (todasAsTags.length === 0) {
                container.innerHTML = '<p class="text-slate-500 text-sm">Nenhuma tag cadastrada. Crie uma abaixo.</p>';
                return;
            }

            todasAsTags.forEach(tag => {
                const estaSelecionada = leadTags.some(lt => lt.id === tag.id);
                const label = document.createElement('label');
                label.className = 'flex items-center gap-2 bg-slate-800 px-3 py-2 rounded border border-slate-700 cursor-pointer hover:border-orange-500 transition-colors';
                label.innerHTML = `
                    <input type="checkbox" ${estaSelecionada ? 'checked' : ''} data-tag-id="${tag.id}" onchange="toggleTagLead(${tag.id}, this.checked)" class="accent-orange-500">
                    <span class="text-sm">${escapeHtml(tag.nome)}</span>
                `;
                container.appendChild(label);
            });
        }

        async function toggleTagLead(tagId, isChecked) {
            if (!leadIdAtual) return;
            const token = localStorage.getItem('token');
            try {
                if (isChecked) {
                    await fetch(`/api/leads/${leadIdAtual}/tags`, {
                        method: 'POST',
                        headers: {
                            'Content-Type': 'application/json',
                            'Authorization': `Bearer ${token}`
                        },
                        body: JSON.stringify({ tag_id: tagId })
                    });
                } else {
                    await fetch(`/api/leads/${leadIdAtual}/tags/${tagId}`, {
                        method: 'DELETE',
                        headers: { 'Authorization': `Bearer ${token}` }
                    });
                }
                await carregarLeads();
            } catch (error) {
                console.error('Erro ao alternar tag:', error);
                alert('Erro ao atualizar tag.');
            }
        }

        document.getElementById('btnCriarTag').addEventListener('click', async () => {
            const nome = document.getElementById('novaTagNome').value;
            const feedback = document.getElementById('feedbackTag');
            if (!nome.trim()) {
                feedback.textContent = 'Digite um nome para a tag.';
                feedback.className = 'mt-2 p-2 rounded text-center text-sm font-bold bg-red-600 text-white';
                return;
            }
            const token = localStorage.getItem('token');
            try {
                const response = await fetch('/api/tags', {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json',
                        'Authorization': `Bearer ${token}`
                    },
                    body: JSON.stringify({ nome })
                });
                const data = await response.json();
                if (!response.ok) throw new Error(data.error || 'Erro ao criar tag');
                
                document.getElementById('novaTagNome').value = '';
                feedback.textContent = 'Tag criada com sucesso!';
                feedback.className = 'mt-2 p-2 rounded text-center text-sm font-bold bg-green-600 text-white';
                
                await carregarTags();
                if (leadIdAtual) await abrirGerenciarTags(leadIdAtual);
                await carregarLeads();
            } catch (error) {
                feedback.textContent = error.message;
                feedback.className = 'mt-2 p-2 rounded text-center text-sm font-bold bg-red-600 text-white';
            }
        });

        document.getElementById('fecharModalTags').addEventListener('click', () => {
            document.getElementById('modalTags').classList.add('hidden');
        });

        document.getElementById('modalTags').addEventListener('click', (e) => {
            if (e.target.id === 'modalTags') {
                document.getElementById('modalTags').classList.add('hidden');
            }
        });

        // --- HISTÓRICO DE INTERAÇÕES ---

        async function abrirHistorico(id) {
            leadIdAtual = id;
            document.getElementById('modalHistorico').classList.remove('hidden');
            await carregarInteracoes(id);
        }

        async function carregarInteracoes(leadId) {
            const token = localStorage.getItem('token');
            const lista = document.getElementById('listaInteracoes');
            lista.innerHTML = '<p class="text-slate-400">Carregando...</p>';

            try {
                const response = await fetch(`/api/interacoes/${leadId}`, {
                    headers: { 'Authorization': `Bearer ${token}` }
                });
                if (!response.ok) throw new Error('Falha ao buscar interações');
                const interacoes = await response.json();

                if (interacoes.length === 0) {
                    lista.innerHTML = '<p class="text-slate-400 text-center py-4">Nenhuma interação registrada ainda.</p>';
                    return;
                }

                lista.innerHTML = '';
                interacoes.forEach(i => {
                    const dataFormatada = new Date(i.data_criacao).toLocaleString('pt-BR');
                    const div = document.createElement('div');
                    div.className = 'bg-slate-800 p-3 rounded-lg mb-2 border border-slate-700';
                    div.innerHTML = `
                        <div class="flex justify-between items-center mb-1">
                            <span class="font-bold text-orange-400">${escapeHtml(i.tipo)}</span>
                            <span class="text-xs text-slate-400">${dataFormatada}</span>
                        </div>
                        <p class="text-slate-200 text-sm">${escapeHtml(i.anotacao)}</p>
                    `;
                    lista.appendChild(div);
                });
            } catch (error) {
                console.error('Erro:', error);
                lista.innerHTML = '<p class="text-red-400">Erro ao carregar interações.</p>';
            }
        }

        document.getElementById('btnSalvarInteracao').addEventListener('click', async () => {
            if (!leadIdAtual) return;
            const token = localStorage.getItem('token');
            const tipo = document.getElementById('tipoInteracao').value;
            const anotacao = document.getElementById('anotacaoInteracao').value;

            if (!anotacao.trim()) {
                alert('Digite uma anotação antes de salvar.');
                return;
            }

            try {
                const response = await fetch('/api/interacoes', {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json',
                        'Authorization': `Bearer ${token}`
                    },
                    body: JSON.stringify({ lead_id: leadIdAtual, tipo, anotacao })
                });
                if (!response.ok) throw new Error('Falha ao salvar interação');
                
                document.getElementById('anotacaoInteracao').value = '';
                await carregarInteracoes(leadIdAtual);
            } catch (error) {
                console.error('Erro:', error);
                alert('Erro ao salvar interação.');
            }
        });

        document.getElementById('fecharModal').addEventListener('click', () => {
            document.getElementById('modalHistorico').classList.add('hidden');
            leadIdAtual = null;
        });

        document.getElementById('modalHistorico').addEventListener('click', (e) => {
            if (e.target.id === 'modalHistorico') {
                document.getElementById('modalHistorico').classList.add('hidden');
                leadIdAtual = null;
            }
        });

                headers: {

                    'Authorization': `Bearer ${token}` 
                },
                body: JSON.stringify({ status: 'atendido', cliente_id: null })
